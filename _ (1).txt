/* ====== CONFIG ====== */
const CONFIG = {
  GAS_URL : "",            // ← วาง URL ของ Google Apps Script Web App ตรงนี้
  ADMIN_PIN : "1234",
  PREFIX_POINT : "MSP69-",     // QR จุด  : MSP69-P01
  PREFIX_TICKET: "MSP69T|"     // QR บัตร : MSP69T|uid|reward|code
};

/* ====== STATE ====== */
const KEY="msp69_state";
let S = JSON.parse(localStorage.getItem(KEY) || "null") || {
  uid:"U"+Date.now().toString(36).toUpperCase()+Math.random().toString(36).slice(2,6).toUpperCase(),
  scanned:[], tickets:[], survey:null, spent:0
};
const save=()=>localStorage.setItem(KEY,JSON.stringify(S));
const earned=()=>S.scanned.reduce((a,b)=>a+b.pts,0);
const balance=()=>earned()-S.spent;
const $=s=>document.querySelector(s), $=s=>[...document.querySelectorAll(s)];

function toast(m){const t=$("#toast");t.textContent=m;t.classList.add("show");clearTimeout(t._);t._=setTimeout(()=>t.classList.remove("show"),2600)}
function openM(id){$(id).classList.add("show")}
function closeM(id){$(id).classList.remove("show")}
$("[data-close]").forEach(b=>b.onclick=e=>{const m=e.target.closest(".modal");m.classList.remove("show");if(m.id==="mdScan")stopScan()});

/* ====== RENDER ====== */
function render(){
  $("#scoreNum").textContent=balance();
  $("#earned").textContent=earned();
  $("#spent").textContent=S.spent;
  $("#scanCount").textContent=`สแกนแล้ว ${S.scanned.length} จุด`;
  $("#barFill").style.width=(S.scanned.length/POINTS.length*100)+"%";
  $("#ptDone").textContent=`${S.scanned.length}/${POINTS.length}`;
  $("#hisCount").textContent=`${S.scanned.length} รายการ`;
  $("#myBal").textContent=`${balance()} คะแนน`;
  renderPoints(); renderHistory(); renderRewards(); renderTickets();
}
let FILTER="all";
function renderPoints(){
  const done=new Set(S.scanned.map(x=>x.id));
  $("#pointList").innerHTML=POINTS.filter(p=>FILTER==="all"||(FILTER==="done")===done.has(p.id))
    .map(p=>`<div class="pt ${done.has(p.id)?"done":""}">
      <div class="pt-no">${done.has(p.id)?"✓":p.no}</div>
      <div class="pt-name">${p.name}</div>
      <div class="pt-pts">${p.pts} คะแนน</div></div>`).join("")||`<div class="empty">ไม่มีรายการ</div>`;
}
function renderHistory(){
  $("#history").innerHTML = S.scanned.length ? [...S.scanned].reverse().map(h=>
    `<div class="his"><div><b>${h.no}. ${h.name}</b><br><small>${h.time}</small></div>
     <div class="pt-pts">+${h.pts}</div></div>`).join("")
   : `<div class="empty"><div class="big">☁️</div>ยังไม่มีประวัติการสแกนเช็คอิน<br><small>เริ่มสแกนเพื่อเก็บคะแนนกันเลย!</small></div>`;
}
function renderRewards(){
  $("#rewardList").innerHTML=REWARDS.map(r=>{
    const ok=balance()>=r.cost;
    return `<div class="rw"><div class="rw-ico" style="background:${r.bg}">${r.ico}</div>
      <div class="rw-info"><b>${r.name}</b><small>ใช้ ${r.cost} คะแนน</small></div>
      <button ${ok?"":"disabled"} data-rw="${r.id}">${ok?"แลกเลย":"คะแนนไม่พอ"}</button></div>`}).join("");
  $("[data-rw]").forEach(b=>b.onclick=()=>startRedeem(b.dataset.rw));
}
function renderTickets(){
  $("#ticketList").innerHTML=S.tickets.length?S.tickets.map(t=>{
    const r=REWARDS.find(x=>x.id===t.reward);
    return `<div class="tk"><div class="rw-ico" style="background:${r.bg};width:44px;height:44px;font-size:22px">${r.ico}</div>
      <div class="rw-info"><b>${r.name}</b><small>รหัส ${t.code} · ${t.time}</small></div>
      <div><span class="tk-s ${t.confirmed?"s-ok":"s-wait"}">${t.confirmed?"รับแล้ว":"รอรับของ"}</span>
      <button style="margin-top:6px" data-tk="${t.code}">แสดง QR</button></div></div>`}).join("")
   : `<div class="empty"><div class="big">🎫</div>ยังไม่มีบัตรรับของรางวัล</div>`;
  $("[data-tk]").forEach(b=>b.onclick=()=>showTicket(b.dataset.tk));
}

/* ====== SCANNER ====== */
let qr=null, MODE="checkin";
function startScan(mode){
  MODE=mode;
  $("#scanTitle").textContent = mode==="admin" ? "สแกนบัตรรับของรางวัล" : "สแกน QR Code เช็คอิน";
  $("#scanHint").textContent  = mode==="admin" ? "หันกล้องไปที่ QR บัตรของผู้ร่วมสนุก" : "หันกล้องไปที่ QR Code ประจำจุด";
  openM("#mdScan");
  qr = new Html5Qrcode("reader");
  qr.start({facingMode:"environment"},{fps:10,qrbox:{width:240,height:240}},
    txt=>{ stopScan(); closeM("#mdScan"); handleCode(txt.trim()); },
    ()=>{}
  ).catch(()=>toast("เปิดกล้องไม่สำเร็จ — ตรวจสอบสิทธิ์กล้อง/ต้องใช้ HTTPS"));
}
function stopScan(){ if(qr){ qr.stop().then(()=>qr.clear()).catch(()=>{}); qr=null; } }
$("#btnScan").onclick=()=>startScan("checkin");
$("#btnAdminScan").onclick=()=>startScan("admin");
$("#btnManual").onclick=()=>{const v=$("#manualCode").value.trim();if(!v)return;stopScan();closeM("#mdScan");$("#manualCode").value="";handleCode(v)};

function handleCode(txt){
  if(MODE==="admin") return adminVerify(txt);
  let code=txt.toUpperCase();
  if(code.includes(CONFIG.PREFIX_POINT)) code=CONFIG.PREFIX_POINT+code.split(CONFIG.PREFIX_POINT)[1].slice(0,3);
  else if(/^P\d{2}$/.test(code)) code=CONFIG.PREFIX_POINT+code;
  const p=POINTS.find(x=>CONFIG.PREFIX_POINT+x.id===code);
  if(!p) return toast("❌ QR Code นี้ไม่ใช่จุดเช็คอินของกิจกรรม");
  if(S.scanned.some(x=>x.id===p.id)) return toast("⚠️ จุดนี้สแกนไปแล้ว ไม่สามารถสแกนซ้ำได้");
  S.scanned.push({id:p.id,no:p.no,name:p.name,pts:p.pts,time:new Date().toLocaleString("th-TH")});
  save(); render(); toast(`✅ ${p.name} +${p.pts} คะแนน`);
}

/* ====== REDEEM + SURVEY ====== */
let pendingReward=null, satVal="";
const SATS=["มากที่สุด","มาก","พอใช้","น้อย","น้อยที่สุด"];
$("#satBox").innerHTML=SATS.map(s=>`<button type="button" data-sat="${s}">${s}</button>`).join("");
$("[data-sat]").forEach(b=>b.onclick=()=>{satVal=b.dataset.sat;$("[data-sat]").forEach(x=>x.classList.remove("on"));b.classList.add("on")});

function startRedeem(id){
  const r=REWARDS.find(x=>x.id===id);
  if(balance()<r.cost) return toast("คะแนนไม่พอ");
  pendingReward=r;
  if(S.survey){ if(confirm(`ยืนยันใช้ ${r.cost} คะแนน แลก "${r.name}" ?`)) doRedeem(); }
  else openM("#mdSurvey");
}
$("#surveyForm").onsubmit=e=>{
  e.preventDefault();
  if(!satVal) return toast("กรุณาเลือกระดับความพึงพอใจ");
  const f=new FormData(e.target);
  S.survey={fullname:f.get("fullname"),age:f.get("age"),address:f.get("address"),
    travel:f.get("travel"),shuttle:f.get("shuttle"),visits:f.get("visits"),
    satisfaction:satVal,suggest:f.get("suggest")||"-",time:new Date().toLocaleString("th-TH")};
  save(); closeM("#mdSurvey"); doRedeem();
};
function doRedeem(){
  const r=pendingReward;
  const code=r.id.slice(0,2).toUpperCase()+Date.now().toString(36).slice(-5).toUpperCase();
  S.spent+=r.cost;
  S.tickets.push({reward:r.id,code,confirmed:false,time:new Date().toLocaleString("th-TH")});
  save(); render(); sendToSheet(r,code); showTicket(code);
  toast(`🎉 แลก ${r.name} สำเร็จ!`);
}
function showTicket(code){
  const t=S.tickets.find(x=>x.code===code), r=REWARDS.find(x=>x.id===t.reward);
  $("#ticketBody").innerHTML=`<div class="rw-ico" style="margin:0 auto 10px;background:${r.bg};width:70px;height:70px;font-size:36px">${r.ico}</div>
    <b style="font-size:19px">${r.name}</b><p class="muted">แสดง QR นี้ให้เจ้าหน้าที่สแกนเพื่อรับของที่ระลึก</p>
    <div id="tkQR" style="display:flex;justify-content:center;background:#fff;padding:16px;border-radius:20px"></div>
    <p style="margin-top:10px;font-weight:700">รหัส ${t.code}</p>
    <span class="tk-s ${t.confirmed?"s-ok":"s-wait"}">${t.confirmed?"รับของแล้ว":"รอรับของ"}</span>`;
  new QRCode($("#tkQR"),{text:`${CONFIG.PREFIX_TICKET}${S.uid}|${t.reward}|${t.code}`,width:200,height:200,correctLevel:QRCode.CorrectLevel.M});
  openM("#mdTicket");
}

/* ====== GOOGLE SHEET ====== */
function sendToSheet(r,code){
  if(!CONFIG.GAS_URL) return console.warn("ยังไม่ได้ตั้งค่า GAS_URL");
  fetch(CONFIG.GAS_URL,{method:"POST",mode:"no-cors",headers:{"Content-Type":"text/plain;charset=utf-8"},
    body:JSON.stringify({action:"redeem",uid:S.uid,reward:r.name,cost:r.cost,code,
      totalScan:S.scanned.length,totalEarn:earned(),
      scannedPoints:S.scanned.map(x=>x.no).join(","),...S.survey})
  }).catch(()=>{});
}

function buildQRGrid(){
  const g=$("#qrGrid"); if(g.dataset.built)return; g.dataset.built=1;
  POINTS.forEach(p=>{
    const d=document.createElement("div");d.className="qr-item";
    const box=document.createElement("div");d.appendChild(box);
    d.insertAdjacentHTML("beforeend",`<div class="n">จุดที่ ${p.no} · ${p.pts} คะแนน<br>${p.name}</div><div class="c">${CONFIG.PREFIX_POINT+p.id}</div>`);
    g.appendChild(d);
    new QRCode(box,{text:CONFIG.PREFIX_POINT+p.id,width:120,height:120,correctLevel:QRCode.CorrectLevel.M});
  });
}

const LOGKEY="msp69_adminlog";
let LOG=JSON.parse(localStorage.getItem(LOGKEY)||"[]");
function adminVerify(txt){
  if(!txt.startsWith(CONFIG.PREFIX_TICKET)) return toast("❌ ไม่ใช่บัตรรับของรางวัล");
  const [,uid,rw,code]=txt.split("|");
  const r=REWARDS.find(x=>x.id===rw)||{name:rw,ico:"🎁"};
  if(LOG.some(l=>l.code===code)) return toast("⚠️ บัตรนี้ถูกยืนยันรับของไปแล้ว");
  $("#verifyResult").innerHTML=`<div class="vr"><b style="font-size:17px">${r.ico} ${r.name}</b>
    <p class="muted">รหัสบัตร: ${code}<br>ผู้เล่น: ${uid}</p>
    <button class="primary" id="btnOk">✅ ยืนยันการรับของที่ระลึก</button></div>`;
  $("#btnOk").onclick=()=>{
    LOG.push({code,reward:r.name,uid,time:new Date().toLocaleString("th-TH")});
    localStorage.setItem(LOGKEY,JSON.stringify(LOG));
    const t=S.tickets.find(x=>x.code===code); if(t){t.confirmed=true;save();render()}
    if(CONFIG.GAS_URL) fetch(CONFIG.GAS_URL,{method:"POST",mode:"no-cors",headers:{"Content-Type":"text/plain;charset=utf-8"},
      body:JSON.stringify({action:"confirm",uid,code,reward:r.name})}).catch(()=>{});
    $("#verifyResult").innerHTML=`<div class="vr" style="text-align:center"><div style="font-size:44px">✅</div><b>ยืนยันสำเร็จ</b><p class="muted">${r.name} · ${code}</p></div>`;
    renderLog(); toast("ยืนยันการรับของเรียบร้อย");
  };
}
function renderLog(){
  $("#adminLog").innerHTML=LOG.length?[...LOG].reverse().map(l=>
    `<div class="his"><div><b>${l.reward}</b><br><small>${l.code} · ${l.time}</small></div><span class="tk-s s-ok">รับแล้ว</span></div>`).join("")
   :`<div class="empty"><div class="big">📋</div>ยังไม่มีการยืนยันรับของ</div>`;
}
